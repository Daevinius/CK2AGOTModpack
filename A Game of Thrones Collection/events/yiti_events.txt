namespace = yiti

# Clear Jungle for Roads
province_event = {
	id = yiti.001
	desc = "EVTDESCyiti.001"
	picture = "GFX_evt_china_famine"


	trigger = {
		culture_group = yi_ti_group
		NOR = {
			has_province_modifier = yiti_road1
			has_province_modifier = yiti_road2
			has_province_modifier = yiti_road3
		}
		owner = { wealth >= 100 }
	}

	mean_time_to_happen = {
		years = 3
	}

	option = {
		name = "EVTOPTAyiti.001"

		owner = { wealth = -50 }

		set_province_flag = build_yiti_road

		add_province_modifier = {
			name = yiti_road1
			duration = -1
		}

		ai_chance = {
			factor = 90
		}
	}
	option = {
		name = "EVTOPTByiti.001"

		ai_chance = {
			factor = 10
		}
	}
}
