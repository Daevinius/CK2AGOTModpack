namespace = new_traits_events

character_event = {
	id = new_traits_events.1

	is_triggered_only = yes # on_startup, on_birth
	hide_window = yes

	only_men = yes

	trigger = {
		OR = {
			trait = wide_hipped
			trait = narrow_hipped
		}
	}

	immediate = {
		remove_trait = wide_hipped
		remove_trait = narrow_hipped
	}
}

character_event = {
	id = new_traits_events.2

	is_triggered_only = yes # on_startup, on_birth
	hide_window = yes

	only_women = yes

	trigger = {
		trait = micro
		trait = well_endowed
	}

	immediate = {
		remove_trait = micro
		remove_trait = well_endowed
	}
}
