character_event = { #Clear flag
	id = speed_dating.0

	# is_triggered_only = yes
	hide_window = yes
	ai = yes

	only_women = yes
	is_married = no
	min_age = 29

	trigger = {
		OR = {
			has_character_flag = northman_match
			has_character_flag = reachman_match
			has_character_flag = westerman_match
			has_character_flag = stormlander_match
			has_character_flag = kingslander_match
			has_character_flag = riverlander_match
			has_character_flag = dorne_match
			has_character_flag = valeman_match
		}
		can_marry_trigger = no
	}

	mean_time_to_happen = {
		months = 1
	}

	immediate = {
		if = {
			limit = { has_character_flag = northman_match }
			clr_character_flag = northman_match
		}
		else_if = {
			limit = { has_character_flag = reachman_match }
			clr_character_flag = reachman_match
		}
		else_if = {
			limit = { has_character_flag = westerman_match }
			clr_character_flag = westerman_match
		}
		else_if = {
			limit = { has_character_flag = stormlander_match }
			clr_character_flag = stormlander_match
		}
		else_if = {
			limit = { has_character_flag = kingslander_match }
			clr_character_flag = kingslander_match
		}
		else_if = {
			limit = { has_character_flag = riverlander_match }
			clr_character_flag = riverlander_match
		}
		else_if = {
			limit = { has_character_flag = dorne_match }
			clr_character_flag = dorne_match
		}
		else = {
			limit = { has_character_flag = valeman_match }
			clr_character_flag = valeman_match
		}
	}

	option = {
		name = OK
	}
}

character_event = { #Noble Landless or Duke+ tier Bachelors of Westeros
	id = speed_dating.9999

	is_triggered_only = yes
	hide_window = yes

	min_age = 16
	max_age = 38
	is_married = no

	ai = yes

	trigger = {
		NOT = { has_character_flag = westeros_bachelor }
		NOT = { at_location = c_the_citadel }
		can_marry_trigger = yes
		is_female = no
		NOT = { has_character_flag = speed_date_block }
		NOT = {
			any_dynasty_member = { ai = no }
		}
		has_dynasty_flag = has_dynamic_coa
		NOR = {
			host = { ai = no }
			any_dynasty_member = { ai = no }
			is_lowborn = yes
			can_marry_trigger = no
			culture = high_valyrian
			has_character_flag = maester_education
		}
		OR = {
			culture_group = valyrian
			culture_group = andal
			culture_group = first_men
			culture_group = dornish
		}
	}

	immediate = {
		set_character_flag = westeros_bachelor
	}

	option = {
		name = OK
	}
}

		name = OK
