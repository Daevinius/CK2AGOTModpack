character_event = {
	id = divorce_events.0
	is_triggered_only = yes
	hide_window = yes

	immediate = {
		ROOT = {
			if = {
				limit = {
					is_married = yes
				}
				set_character_flag = no_wedding_event
				any_spouse = {
					set_character_flag = no_wedding_event
					remove_spouse = ROOT
					clr_character_flag = no_wedding_event
				}
				clr_character_flag = no_wedding_event
			}
			else_if = {
				limit = {
					is_betrothed = yes
				}
				betrothed = {
					break_betrothal = ROOT
				}
			}
			if = {
				limit = { is_female = no }
				prestige = 150
			}
		}
	}
}

character_event = { #No Lowborns or weird cultures
	id = divorce_events.1

	is_triggered_only = yes
	hide_window = yes
	ai = yes

	trigger = {
		ROOT = {
			higher_tier_than = BARON
			OR = {
				culture = western_valyrian
				culture = high_valyrian
				culture_group = first_men
				culture_group = andal
				culture_group = dornish
			}
			has_dynasty_flag = has_dynamic_coa
		}
		FROM = {
			is_lowborn = no
			OR = {
				culture = western_valyrian
				culture = high_valyrian
				culture_group = first_men
				culture_group = andal
				culture_group = dornish
			}
		}
		new_character = {
			OR = {
				is_lowborn = yes
				culture_group = jogos_nhai_group
				culture = brindlemen
				culture = gogossosi
			}
			OR = {
				#Landless and not in player's court
				AND = {
					NOT = { is_ruler = yes }
					liege = { ai = no }
				}
				#Landed and is not player
				AND = {
					is_ruler = yes
					ai = yes
				}
			}
		}
	}

	immediate = {
		FROM = {
			character_event = { id = divorce_events.0 days = 1 }
		}
	}
}
