# Armors Artifacts
character_event = {
	id = mbs_artifacts.001
	desc = EVTDESCmbs_artifacts.001
	picture = GFX_knighted

	is_triggered_only = yes # on_startup, on_yearly_pulse

	trigger = {
		trait = kingsguard
		NOT = { any_artifact = { has_artifact_flag = kingsguard_armour } }
	}

	option = {
		name = EVTOPTAmbs_artifacts.001 #

		if = {
			limit = { dynasty = 190 }
			add_artifact = jaime_lannister_armour
		}
		else = {
			add_artifact = kingsguard_armour
		}
	}
}
